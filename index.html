function loadTags(i) { if (i >= songs.length) { buildPlaylist(); playSong(); return; } const fileUrl = `https://raw.githubusercontent.com/spitraklo/-Sounds/main/${songs[i].file}`; jsmediatags.read(fileUrl, { onSuccess: function(tag) { if (tag.tags.artist) songs[i].artist = tag.tags.artist; loadTags(i + 1); }, onError: function() { loadTags(i + 1); } }); }
